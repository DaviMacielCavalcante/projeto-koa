# Autenticação por SMS OTP — Plano de Implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implementar autenticação via Firebase Auth (phone OTP/SMS) com React Navigation e session timeout de 1h em background / 3h de inatividade.

**Architecture:** `AuthContext` centraliza o estado de auth, lógica de session timeout e acesso ao `expo-secure-store`. `App.tsx` monta o `NavigationContainer` e alterna entre `AuthStack` (pré-login) e `AppStack` (pós-login) com base no `user` do contexto.

**Tech Stack:** React Native + Expo SDK 54, TypeScript, `@react-native-firebase/auth`, `@react-navigation/stack`, `expo-secure-store`, Jest + jest-expo + @testing-library/react-native

```ts
// apps/mobile/src/utils/phoneMask.ts
export function applyPhoneMask(raw: string): string {
  const digits = raw.replace(/\D/g, '').slice(0, 11);
  if (digits.length === 0) return '';
  if (digits.length <= 2) return `(${digits}`;
  if (digits.length <= 7) return `(${digits.slice(0, 2)}) ${digits.slice(2)}`;
  return `(${digits.slice(0, 2)}) ${digits.slice(2, 7)}-${digits.slice(7)}`;
}

export function stripPhoneMask(masked: string): string {
  return masked.replace(/\D/g, '');
}
```

```tsx
// apps/mobile/src/contexts/AuthContext.tsx
import React, { createContext, useEffect, useRef, useState } from 'react';
import { AppState, AppStateStatus } from 'react-native';
import auth, { FirebaseAuthTypes } from '@react-native-firebase/auth';
import * as SecureStore from 'expo-secure-store';

const BACKGROUND_LIMIT_MS = 60 * 60 * 1000;     // 1 hora
const INACTIVITY_LIMIT_MS = 3 * 60 * 60 * 1000; // 3 horas
const BACKGROUND_TS_KEY = 'background_timestamp';

interface AuthContextValue {
  user: FirebaseAuthTypes.User | null;
  isLoading: boolean;
  signOut: () => Promise<void>;
  recordActivity: () => void;
}

export const AuthContext = createContext<AuthContextValue | null>(null);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<FirebaseAuthTypes.User | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const lastActivityRef = useRef<number>(Date.now());
  const inactivityTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  function clearInactivityTimer() {
    if (inactivityTimerRef.current !== null) {
      clearTimeout(inactivityTimerRef.current);
      inactivityTimerRef.current = null;
    }
  }

  async function doSignOut() {
    clearInactivityTimer();
    await SecureStore.deleteItemAsync(BACKGROUND_TS_KEY);
    await auth().signOut();
  }

  function startInactivityTimer(remainingMs: number = INACTIVITY_LIMIT_MS) {
    clearInactivityTimer();
    inactivityTimerRef.current = setTimeout(() => { doSignOut(); }, remainingMs);
  }

  function recordActivity() {
    lastActivityRef.current = Date.now();
    startInactivityTimer();
  }

  useEffect(() => {
    const unsubscribe = auth().onAuthStateChanged((firebaseUser) => {
      setUser(firebaseUser);
      setIsLoading(false);
      if (firebaseUser) {
        lastActivityRef.current = Date.now();
        startInactivityTimer();
      } else {
        clearInactivityTimer();
      }
    });
    return unsubscribe;
  }, []);

  useEffect(() => {
    if (!user) return;

    const subscription = AppState.addEventListener('change', async (state: AppStateStatus) => {
      if (state === 'background' || state === 'inactive') {
        clearInactivityTimer();
        await SecureStore.setItemAsync(BACKGROUND_TS_KEY, Date.now().toString());
      } else if (state === 'active') {
        const stored = await SecureStore.getItemAsync(BACKGROUND_TS_KEY);
        if (stored) {
          const backgroundDuration = Date.now() - parseInt(stored, 10);
          await SecureStore.deleteItemAsync(BACKGROUND_TS_KEY);
          if (backgroundDuration >= BACKGROUND_LIMIT_MS) {
            await doSignOut();
            return;
          }
        }
        const elapsed = Date.now() - lastActivityRef.current;
        if (elapsed >= INACTIVITY_LIMIT_MS) {
          await doSignOut();
        } else {
          startInactivityTimer(INACTIVITY_LIMIT_MS - elapsed);
        }
      }
    });

    return () => subscription.remove();
  }, [user]);

  return (
    <AuthContext.Provider value={{ user, isLoading, signOut: doSignOut, recordActivity }}>
      {children}
    </AuthContext.Provider>
  );
}
```

```tsx
  Platform,
  StyleSheet,
  Text,
  TextInput,
  TouchableOpacity,
  View,

import { useNavigation } from '@react-navigation/native';
import type { StackNavigationProp } from '@react-navigation/stack';
import type { AuthStackParamList } from '../../navigation/AuthStack';
import { applyPhoneMask, stripPhoneMask } from '../../utils/phoneMask';

type Nav = StackNavigationProp<AuthStackParamList, 'Phone'>;

export default function PhoneScreen() {
  const navigation = useNavigation<Nav>();
  const [phoneDisplay, setPhoneDisplay] = useState('');
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');

  function handleChangePhone(text: string) {
    setError('');
    setPhoneDisplay(applyPhoneMask(text));

  async function handleSendCode() {
    const digits = stripPhoneMask(phoneDisplay);
    if (digits.length < 10) {
      setError('Número de telefone inválido');
      // TODO audio: error_invalid_number.mp3
      return;

    setError('');

      const confirmation = await auth().signInWithPhoneNumber(phoneWithCountry);
      navigation.navigate('Otp', {
        verificationId: confirmation.verificationId,
        phoneNumber: phoneWithCountry,
      });
    } catch (e: any) {
      if (e.code === 'auth/invalid-phone-number') {
        setError('Número de telefone inválido');
        // TODO audio: error_invalid_number.mp3
      } else if (e.code === 'auth/network-request-failed') {
        setError('Não foi possível enviar o SMS. Verifique sua conexão.');
        // TODO audio: error_no_signal.mp3
      } else {
        setError('Erro ao enviar o código. Tente novamente.');
        // TODO audio: error_generic.mp3
      }
    } finally {
      setLoading(false);
    }
  }

  return (

        placeholder="(91) 99999-9999"

        disabled={loading}
        accessibilityLabel="Receber código por SMS"

        {loading ? (
          <ActivityIndicator color="#fff" />
        ) : (
          <Text style={styles.buttonText}>Receber código</Text>
```

```tsx
import auth from '@react-native-firebase/auth';
import { useRoute } from '@react-navigation/native';
import type { RouteProp } from '@react-navigation/native';
import type { AuthStackParamList } from '../../navigation/AuthStack';

type OtpRoute = RouteProp<AuthStackParamList, 'Otp'>;

export default function OtpScreen() {
  const route = useRoute<OtpRoute>();
  const { phoneNumber } = route.params;
  const [verificationId, setVerificationId] = useState(route.params.verificationId);
  const [digits, setDigits] = useState<string[]>(['', '', '', '', '', '']);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');
  const [secondsLeft, setSecondsLeft] = useState(60);
  const [canResend, setCanResend] = useState(false);
  const inputRefs = useRef<Array<TextInput | null>>([]);

  useEffect(() => {
    if (secondsLeft === 0) {
      setCanResend(true);
      return;
    }
    const timer = setTimeout(() => setSecondsLeft((s) => s - 1), 1000);
    return () => clearTimeout(timer);
  }, [secondsLeft]);

  function handleDigitChange(index: number, value: string) {
    const digit = value.replace(/\D/g, '').slice(-1);
    const next = [...digits];
    next[index] = digit;
    setDigits(next);
    setError('');
    if (digit && index < 5) {
      inputRefs.current[index + 1]?.focus();
    }
  }

  function handleKeyPress(index: number, key: string) {
    if (key === 'Backspace' && !digits[index] && index > 0) {
      inputRefs.current[index - 1]?.focus();
    }
  }

  async function handleConfirm() {
    const code = digits.join('');
    if (code.length < 6) {
      setError('Por favor, insira o código completo');
      return;
    }
    setLoading(true);
    setError('');
    try {
      const credential = auth.PhoneAuthProvider.credential(verificationId, code);
      await auth().signInWithCredential(credential);
      // onAuthStateChanged no AuthContext detecta o usuário → App.tsx troca para AppStack
    } catch (e: any) {
      if (e.code === 'auth/invalid-verification-code') {
        setError('Código incorreto. Tente novamente.');
        // TODO audio: error_wrong_code.mp3
      } else if (e.code === 'auth/code-expired') {
        setError('O código expirou. Volte e peça um novo.');
        // TODO audio: error_expired_code.mp3
      } else {
        setError('Erro ao confirmar. Tente novamente.');
        // TODO audio: error_generic.mp3
      }
    } finally {
      setLoading(false);
    }
  }

  async function handleResend() {
    try {
      const confirmation = await auth().signInWithPhoneNumber(phoneNumber);
      setVerificationId(confirmation.verificationId);
      setDigits(['', '', '', '', '', '']);
      setSecondsLeft(60);
      setCanResend(false);
      setError('');
      inputRefs.current[0]?.focus();
    } catch {
      setError('Erro ao reenviar o código. Tente novamente.');
    }
  }

  return (
    <KeyboardAvoidingView
      style={styles.container}
      behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
    >
      {/* TODO audio: otp_instructions.mp3 (autoPlay na primeira visita) */}
      <Text style={styles.title}>Digite o código recebido por SMS</Text>
      <View style={styles.codeRow}>
        {digits.map((digit, i) => (
          <TextInput
            key={i}
            ref={(el) => { inputRefs.current[i] = el; }}
            style={styles.digitInput}
            value={digit}
            onChangeText={(v) => handleDigitChange(i, v)}
            onKeyPress={({ nativeEvent }) => handleKeyPress(i, nativeEvent.key)}
            keyboardType="number-pad"
            maxLength={1}
            editable={!loading}
            selectTextOnFocus
          />
        ))}
      </View>
      {error ? <Text style={styles.error}>{error}</Text> : null}
      <TouchableOpacity
        style={[styles.button, loading && styles.buttonDisabled]}
        onPress={handleConfirm}
        disabled={loading}
        accessibilityLabel="Confirmar código"
      >
        {loading ? (
          <ActivityIndicator color="#fff" />
        ) : (
          <Text style={styles.buttonText}>Confirmar</Text>
        )}
      </TouchableOpacity>
      {canResend ? (
        <TouchableOpacity
          style={styles.resendButton}
          onPress={handleResend}
          accessibilityLabel="Reenviar código por SMS"
        >
          <Text style={styles.resendText}>Reenviar código</Text>
        </TouchableOpacity>
      ) : (
        <Text style={styles.timer}>Reenviar em {secondsLeft}s</Text>
      )}
    </KeyboardAvoidingView>
  );
}

const styles = StyleSheet.create({

});
```
